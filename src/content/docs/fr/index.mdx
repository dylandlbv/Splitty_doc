import { Card, CardGrid } from '@astrojs/starlight/components';

## Pourquoi Splitty ?

<CardGrid stagger>
	<Card title="Multi-foyers" icon="seti:folder">
		Chaque foyer a ses membres, ses revenus et ses dépenses. Le créateur du foyer est administrateur et invite les autres membres par email.
	</Card>
	<Card title="Répartition au choix" icon="random">
		Dépense commune répartie de façon égalitaire, ou proportionnelle aux revenus déclarés du mois.
	</Card>
	<Card title="Remboursements optimisés" icon="approve-check">
		Pour chaque mois, Splitty calcule les soldes de chacun et suggère les virements (méthode gloutonne) qui minimisent le nombre de transferts.
	</Card>
	<Card title="Zéro erreur d'arrondi" icon="seti:db">
		Tous les montants sont stockés en centimes (bigint) : aucun calcul en virgule flottante.
	</Card>
</CardGrid>
